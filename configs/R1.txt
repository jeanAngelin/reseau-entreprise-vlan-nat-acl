! ============================================================
! R1 - Routeur Cisco 2911 (passerelle de l'entreprise)
! Inter-VLAN (router-on-a-stick), DHCP, NAT/PAT, ACL
! G0/0 -> trunk vers SW1 (Fa0/24) ; G0/1 -> FAI (203.0.113.0/30)
! Configuration a coller dans le CLI (mode enable)
! ============================================================
enable
configure terminal
hostname R1

! --- Etape 2 : routage inter-VLAN ---
interface g0/0
 no shutdown
interface g0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
interface g0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
interface g0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
exit

! --- Etape 3 : DHCP (un pool par VLAN) ---
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp excluded-address 192.168.30.1 192.168.30.10
ip dhcp pool COMPTA
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.20.10
ip dhcp pool INFO
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.20.10
ip dhcp pool DIRECTION
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.20.10
exit

! --- Etape 5 : NAT/PAT et route par defaut ---
interface g0/1
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
exit
interface g0/0.10
 ip nat inside
interface g0/0.20
 ip nat inside
interface g0/0.30
 ip nat inside
exit
access-list 1 permit 192.168.0.0 0.0.255.255
ip nat inside source list 1 interface g0/1 overload
ip route 0.0.0.0 0.0.0.0 203.0.113.1

! --- Etape 6 : ACL (Comptabilite -> Direction refusee) ---
ip access-list extended COMPTA_IN
 deny ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
 permit ip any any
exit
interface g0/0.10
 ip access-group COMPTA_IN in
end
write memory

! Verifications :
! show ip interface brief
! show ip dhcp binding
! show ip nat translations
! show access-lists
